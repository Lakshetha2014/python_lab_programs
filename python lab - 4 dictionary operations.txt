# Dictionary Operations in Python

# 1. Creating a dictionary
my_dict = {"name": "Alice", "age": 25, "city": "Chennai"}
print("Original Dictionary:", my_dict)

# 2. Accessing elements
print("Name:", my_dict["name"])          # Using key
print("Age:", my_dict.get("age"))        # Using get()

# 3. Adding a new element
my_dict["country"] = "India"
print("After Adding Country:", my_dict)

# 4. Updating an element
my_dict["age"] = 26
print("After Updating Age:", my_dict)

# 5. Deleting elements
del my_dict["city"]                       # Delete specific key
print("After Deleting City:", my_dict)

removed_value = my_dict.pop("age")        # Remove and return value
print("Removed Age Value:", removed_value)
print("After Removing Age:", my_dict)

# Clear all elements
temp_dict = my_dict.copy()                # Copy to temp_dict
print("temp_dict is:", temp_dict)
temp_dict.clear()
print("After Clearing Dictionary:", temp_dict)

# 6. Iterating through dictionary
my_dict = {"name": "Alice", "age": 25, "city": "Chennai"}
print("\nKeys:")
for key in my_dict.keys():
    print(key)

print("\nValues:")
for value in my_dict.values():
    print(value)

print("\nKey-Value Pairs:")
for key, value in my_dict.items():
    print(key, ":", value)

# 7. Checking if a key exists
if "name" in my_dict:
    print("\n'name' key exists in dictionary")

# 8. Dictionary length
print("Length of Dictionary:", len(my_dict))
