my_list = [5, 10, 15, 20]
my_list.append(25) # Add at end
print("after append my_list is",my_list)
my_list.insert(2, 12) # Insert at index 2
print("after insert operation my_list is",my_list)                 
my_list.extend([30, 35]) # Extend list
print("after extend operation my_list is",my_list)   
my_list.remove(10)# Remove value 10
print(" after remove operation my_list is",my_list)   
my_list.pop(1) # Remove element at index 1
print("after remove operation my_list is",my_list)   
my_list.reverse() # Reverse list
print("my_list is",my_list)   
my_list.sort() # Sort list
print("my_list is",my_list)   
print(len(my_list)) # Length
print(15 in my_list)# Membership
for i in my_list:
    print(i)
